cmake_minimum_required(VERSION 3.18)
project(csplendor_final_common_adapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT IS_DIRECTORY "${ENGINE_ROOT}/src" OR NOT EXISTS "${COMMON_BENCHMARK_SOURCE}")
  message(FATAL_ERROR "Explicit engine tree and generated common benchmark required")
endif()
# These verifier names did not exist on old main. Record explicit OFF, not an
# absent value; never suppress a verifier mismatch in the existing runner.
foreach(name CSPLENDOR_VERIFY_INCREMENTAL_HASH CSPLENDOR_VERIFY_REVEAL_SEARCH_STATE
             CSPLENDOR_VERIFY_REVEAL_SCORE_ORDER CSPLENDOR_VERIFY_SOLVER_ROLLBACK)
  option(${name} "Enable an engine verifier (not a deployment measurement)" OFF)
endforeach()
option(CSPLENDOR_BUILD_ENGINE_BENCHMARK "Build engine-owned benchmark" OFF)
option(CSPLENDOR_PERF_INSTRUMENTATION "Engine performance instrumentation" OFF)
if(CSPLENDOR_PERF_INSTRUMENTATION)
  message(FATAL_ERROR "Common adapter is deployment-only; main has no PERF hooks")
endif()
add_subdirectory("${ENGINE_ROOT}" engine)
add_executable(final_common_benchmark "${COMMON_BENCHMARK_SOURCE}")
target_include_directories(final_common_benchmark PRIVATE "${COMMON_BENCHMARK_COMPAT}")
target_link_libraries(final_common_benchmark PRIVATE csplendor_core)
if(CSPLENDOR_ENABLE_LTO)
  set_property(TARGET final_common_benchmark PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
